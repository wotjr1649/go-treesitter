================================================================================
Hangul declaration value
================================================================================

a { b: 한글 }

--------------------------------------------------------------------------------

(stylesheet
  (rule_set
    (selectors (tag_name))
    (block
      (declaration (property_name) (plain_value)))))

================================================================================
Non-Latin-1 declaration values with terminators
================================================================================

a { animation-name: -한글 !important; font-family: 𠀀글; }

--------------------------------------------------------------------------------

(stylesheet
  (rule_set
    (selectors (tag_name))
    (block
      (declaration (property_name) (plain_value) (important))
      (declaration (property_name) (plain_value)))))

================================================================================
A numeric unit is not a separate identifier value
================================================================================

a { b: 1한 }

--------------------------------------------------------------------------------

(stylesheet
  (rule_set
    (selectors (tag_name))
    (block
      (declaration (property_name) (ERROR) (plain_value)))))

================================================================================
A function name is not a separate identifier value
================================================================================

a { b: 한글(abc) }

--------------------------------------------------------------------------------

(stylesheet
  (rule_set
    (selectors (tag_name))
    (block
      (declaration (property_name) (plain_value))
      (ERROR))))

================================================================================
Existing values and delimiters
================================================================================

a { b: a한, é, 12px, var(--한글), "한글"; }

--------------------------------------------------------------------------------

(stylesheet
  (rule_set
    (selectors (tag_name))
    (block
      (declaration (property_name) (plain_value) (plain_value) (integer_value (unit)) (call_expression (function_name) (arguments (plain_value))) (string_value (string_content))))))
